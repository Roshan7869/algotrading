import chromadb
from chromadb.config import Settings
from strategy_db.config import DB_DIR, COLLECTION_NAME
from collections import Counter

client = chromadb.PersistentClient(path=DB_DIR, settings=Settings(anonymized_tel
emetry=False))
collection = client.get_collection(name=COLLECTION_NAME)
all_data = collection.get(include=['metadatas'])

# Count by setup_type
setup_types = Counter()
# Count by market_condition
market_conds = Counter()

for m in all_data['metadatas']:
    st = m.get('setup_type', 'unknown')
    mc = m.get('market_condition', 'unknown')
    setup_types[st] += 1
    market_conds[mc] += 1

print('=== COUNT BY SETUP-TYPE ===')
for k, v in sorted(setup_types.items()):
    print(f'  {k}: {v}')

print()
print('=== COUNT BY MARKET-CONDITION ===')
for k, v in sorted(market_conds.items()):
    print(f'  {k}: {v}')

print(f'\nTotal: {sum(setup_types.values())}')
